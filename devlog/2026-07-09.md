import { z } from "zod";

export const optionalNumber = (
  min: number,
  max: number,
  msgMin: string,
  msgMax: string,
) =>
  z
    .union([z.string(), z.number()])
    .optional()
    .transform((v) => (v == null ? "" : String(v).trim()))
    .transform((v) => (v === "" ? undefined : Number(v)))
    .refine((v) => v === undefined || Number.isFinite(v), {
      message: "数字を入力して下さい",
    })
    .refine((v) => v === undefined || v >= min, { message: msgMin })
    .refine((v) => v === undefined || v <= max, { message: msgMax });

export const makePatientSchemaPartial = (usedRooms: number[]) => {
  return z
    .object({
      name: z
        .string()
        .trim()
        .min(1, "氏名は必須です")
        .refine((v) => !/\d/.test(v), {
          message: "氏名に数字を含めることはできません",
        }),
      room: optionalNumber(1, 999, "部屋番号は1以上", "部屋番号は999以下"),
      age: optionalNumber(0, 150, "年齢は0以上", "年齢は150以下"),
      disease: z.string().trim().optional(),
      history: z.string().trim().optional(),
      progress: z.string().trim().optional(),
    })
    .superRefine((data, ctx) => {
      if (data.room == null) return;

      if (usedRooms.includes(data.room)) {
        ctx.addIssue({
          code: z.ZodIssueCode.custom,
          message: "この部屋番号は既に使用されています",
          path: ["room"],
        });
      }
    });
};

    const result = schema.safeParse(testCase.input);

    const firstPath = result.success
      ? ""
      : (result.error.issues[0]?.path ?? []).join(".");

    return {
      id: testCase.id,
      label: testCase.label,
      expected: testCase.expectValid ? "valid" : "invalid",
      actual: result.success ? "valid" : "invalid",
      ok:
        result.success === testCase.expectValid &&
        (testCase.expectErrorPath == null ||
          firstPath === testCase.expectErrorPath),

      firstErrorMessage: result.success

## 理解したこと

### 1. `optionalNumber` の引数に型を付けた

JavaScript では以下のように書いていた。

```ts
export const optionalNumber = (min, max, msgMin, msgMax) =>
```

TypeScript では、引数の型を明示する。

```ts
export const optionalNumber = (
  min: number,
  max: number,
  msgMin: string,
  msgMax: string,
) =>
```

`min` と `max` は数値なので `number`。
`msgMin` と `msgMax` はエラーメッセージなので `string`。

---

### 2. `usedRooms` は number の配列

`makePatientSchemaPartial` の引数 `usedRooms` は、使用済みの部屋番号を表す。

```ts
[101, 102]
```

のような数値の配列なので、型は `number[]` になる。

```ts
export const makePatientSchemaPartial = (usedRooms: number[]) => {
```

---

### 3. `PatientValidationCase` 型を作った

テストケース1件分の形を TypeScript に教えるために、型を作成した。

```ts
type PatientValidationCase = {
  id: string;
  label: string;
  usedRooms: number[];
  input: {
    name: string;
    room: string;
    age: string;
    disease: string;
    history: string;
    progress: string;
  };
  expectValid: boolean;
  expectErrorPath?: string;
  expectErrorMessage?: string;
};
```

`expectErrorPath` と `expectErrorMessage` は、正常入力のケースでは不要なので `?` を付けて任意プロパティにした。

---

### 4. `createPatientValidationCases` の戻り値に型を付けた

```ts
export const createPatientValidationCases = (): PatientValidationCase[] => [
```

これは、

```txt
PatientValidationCase 型のデータが複数入った配列を返す
```

という意味。

---

### 5. `window.runPatientValidationCases` の意味

このコードは、開発中だけブラウザのコンソールから `runPatientValidationCases()` を実行できるようにするためのもの。

```ts
if (import.meta.env?.DEV && typeof window !== "undefined") {
  (
    window as typeof window & {
      runPatientValidationCases: typeof runPatientValidationCases;
    }
  ).runPatientValidationCases = runPatientValidationCases;
}
```

実質的にやっていることはこれ。

```ts
window.runPatientValidationCases = runPatientValidationCases;
```

ただし、TypeScript では標準の `window` に `runPatientValidationCases` というプロパティは存在しない。

そのため、

```ts
window as typeof window & {
  runPatientValidationCases: typeof runPatientValidationCases;
}
```

と書いて、

```txt
この window は、普通の window に加えて
runPatientValidationCases も持っているものとして扱ってください
```

と TypeScript に教えている。

---

### 6. `import.meta.env?.DEV` の意味

```ts
import.meta.env?.DEV
```

は、Vite の開発環境かどうかを確認している。

```txt
npm run dev 中なら true
本番ビルドなら false
```

というイメージ。

つまり、このコードは本番ではなく、開発中だけ動く。

---

### 7. `typeof window !== "undefined"` の意味

```ts
typeof window !== "undefined"
```

は、ブラウザ環境かどうかを確認している。

ブラウザには `window` がある。
Node.js やサーバー側には基本的に `window` がない。

そのため、安全に確認してから `window` を使っている。
