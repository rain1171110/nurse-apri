### PatientListのテスト

```tsx
// @vitest-environment jsdom

import { afterEach, describe, expect, it, vi } from "vitest";
import { cleanup, render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { MemoryRouter } from "react-router-dom";
import PatientList from "./PatientList";

afterEach(() => {
  cleanup();
});

describe("PatientList", () => {
  it("氏名で検索すると一致する患者だけを表示する", async () => {
    const user = userEvent.setup();

    const patients = [
      { id: "patient1", name: "田中", room: 101 },
      { id: "patient2", name: "山本", room: 202 },
    ];

    render(
      <MemoryRouter>
        <PatientList
          patients={patients}
          isLoading={false}
          onErrorsChange={vi.fn()}
          addPatient={vi.fn().mockResolvedValue(undefined)}
        />
      </MemoryRouter>,
    );

    const searchInput = screen.getByLabelText("患者検索");
    await user.type(searchInput, "田中");

    expect(
      screen.getByRole("heading", { name: "101号室 田中" }),
    ).toBeInTheDocument();

    expect(
      screen.queryByRole("heading", { name: "202号室 山本" }),
    ).not.toBeInTheDocument();
  });

  it("部屋番号で検索すると一致する患者だけを表示する", async () => {
    const user = userEvent.setup();

    const patients = [
      { id: "patient1", name: "田中", room: 101 },
      { id: "patient2", name: "山本", room: 202 },
    ];

    render(
      <MemoryRouter>
        <PatientList
          patients={patients}
          isLoading={false}
          onErrorsChange={vi.fn()}
          addPatient={vi.fn().mockResolvedValue(undefined)}
        />
      </MemoryRouter>,
    );

    const searchInput = screen.getByLabelText("患者検索");
    await user.type(searchInput, "101");

    expect(
      screen.getByRole("heading", { name: "101号室 田中" }),
    ).toBeInTheDocument();

    expect(
      screen.queryByRole("heading", { name: "202号室 山本" }),
    ).not.toBeInTheDocument();
  });
});
```

## 4. 気づき・理解したこと

- `String(patient.room)` は、数値の部屋番号を検索用の文字列に変換する。元の部屋番号は数値のまま。
- `includes` は部分一致を判定する。
- `includes("")` は `true` になるため、検索欄を空にすると全患者が検索結果に入る。
- `||` は、氏名・部屋番号のどちらか一方が一致すれば条件を満たす。
- `filter` で絞り込み、`map` で表示要素を作る。
- 存在する要素の確認には `getByRole`、存在しない要素の確認には `queryByRole` を使う。
- `await user.type(...)` で入力操作の完了を待つ。
- `MemoryRouter` で、`navigate` を使うコンポーネントをテストできる。
