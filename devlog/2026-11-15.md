#### 2. ログインAPIでトークンを取得
```typescript
// src/api/authApi.ts
export const loginApi = async (receiveLogin: LoginData): Promise<void> => {
  const response = await fetch(`${API_BASE}/auth/login`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(receiveLogin),
    credentials: "include",  // クッキーを自動で送受信
  });
  if (!response.ok) {
    await throwApiError(response);
  }
};
```

#### 3. ログイン成功後、患者データを取得
```typescript
// src/App.tsx
const handleLoginSuccess = async (): Promise<void> => {
  await loadAppData();  // 患者データを取得
};

const loadAppData = async (): Promise<void> => {
  setLoading(true);
  setApiError("");
  try {
    const data = await fetchAppData();  // APIを呼ぶ
    setIsLoggedIn(true);
    setAppData({
      patients: Array.isArray(data.patients) ? data.patients : [],
      records: Array.isArray(data.records) ? data.records : [],
    });
  } catch (error) {
    if (error instanceof Error && error.message === "API error:401") {
      setIsLoggedIn(false);
    }
  } finally {
    setLoading(false);
  }
};
```

#### 4. APIでサーバーからデータを取得
```typescript
// src/api/appDataApi.ts
export const fetchAppData = async (): Promise<AppData> => {
  const response = await fetch(`${API_BASE}/data`, { 
    credentials: "include"  // クッキーに入ったトークンが自動で送信される
  });
  if (!response.ok) {
    throw new Error(`API error:${response.status}`);
  }
  const data: AppData = await response.json();
  return data;
};
```

#### 5. サーバーでトークン検証とデータ取得
```typescript
// server/auth.ts - 認証ミドルウェア
export const requireAuth = (
  req: Request,
  res: Response,
  next: NextFunction,
) => {
  try {
    const token = req.cookies.token;  // クッキーからトークンを取り出す
    if (!token) {
      res.status(401).json({ error: "認証が必要です" });
      return;
    }
    const decoded = jwt.verify(token, JWT_SECRET);  // JWTを検証
    if (typeof decoded === "string" || typeof decoded.userId !== "string") {
      res.status(401).json({ error: "トークン内容が無効です" });
      return;
    }
    req.userId = decoded.userId;  // リクエストにユーザーIDを設定
    next();
  } catch (error) {
    res.status(401).json({ error: "トークンが無効または期限切れです" });
  }
};
```

#### 6. そのユーザーの患者データだけを取得
```typescript
// server/index.ts
app.get("/api/data", requireAuth, async (req, res) => {
  const userId = req.userId;
  if (!userId) {
    res.status(401).json({ error: "認証が必要です" });
    return;
  }
  try {
    // userId でフィルタして、そのユーザーの患者データだけ取得
    const patients = await prisma.patient.findMany({
      where: {
        userId,  // ここが重要！この行で他ユーザーのデータが混ざらない
      },
    });

    // 患者に関連する看護記録も同じく userId でフィルタ
    const records = await prisma.nursingRecord.findMany({
      where: {
        patient: {
          userId,  // ネストされた条件でもユーザーIDをチェック
        },
      },
      include: {
        vitals: true,
      },
    });

    const data = { patients, records };
    res.json(data);
  } catch (error) {
    res.status(500).json({ error: "データ読み込み失敗" });
  }
});
```

### 解説メモ（会話からそのまま転記）

- 質問・テーマ：「ログイン後の患者データ取得はどうやって動いている？」
- 解説（原文のまま）：
  - ログイン → トークン生成 → クッキーに保存
  - 患者データ取得APIをコールするとき、クッキーが自動で一緒に送信される（`credentials: "include"`）
  - サーバーは認証ミドルウェア `requireAuth` で、クッキーからトークンを取り出してJWT検証
  - トークンが有効ならユーザーIDを取り出す
  - その後のエンドポイント `/api/data` で、`userId` を条件にしてデータベースをクエリ
  - 結果として、そのユーザーの患者データだけが返される

### 自分の理解

- **このコードの役割**
  - ユーザーがログインしたときに、そのユーザー固有の患者データをサーバーから取得して、アプリに表示すること
  - 同時に、他のユーザーの患者データが漏洩しないように、サーバー側で`userId`で厳しくフィルタすること

- **入力と出力**
  - 入力：メールとパスワード
  - 出力：そのユーザーの患者リスト（患者情報と看護記録）

- **重要な処理の流れ**
  1. ログインフォーム送信 → ログインAPI呼び出し
  2. サーバーが認証 → トークン生成 → クッキーに保存
  3. フロントで `loadAppData()` 実行 → `fetchAppData()` でサーバー呼び出し
  4. クッキーが自動で送信される
  5. サーバーの `requireAuth` がトークン検証 → `userId` 取り出す
  6. `/api/data` エンドポイントで、`userId` を条件にデータ取得
  7. フロント側で `setAppData()` で状態更新 → 画面に患者リスト表示

- **修正後の変化**

- **理解したこと**
  - **クッキーと `credentials: "include"`** の組み合わせで、APIリクエストごとに自動的にトークンが送信される
  - **サーバー側の `userId` フィルタリング** が、セキュリティの要
  - **requireAuth ミドルウェア** で一度検証するだけでなく、各エンドポイントでも `userId` を確認している
  - Prismaの `where` 句で `userId` 条件を付ければ、他ユーザーのデータは絶対に取得できない設計になっている
